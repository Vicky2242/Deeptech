select CUSTOMER_ID,ISOLVE_RECIVED_DATE,CUSTOMER_STATUS,CKYC_NUMBER,RECEVIED_DATE from (
SELECT distinct  H.CUSTOMER_REFERENCE_NUMBER AS CUSTOMER_ID,CONVERT(VARCHAR(10), H.IS_PROCESS_DATE, 105)
AS ISOLVE_RECIVED_DATE,IRM_RESPONSEDESC AS CUSTOMER_STATUS,CM_CKYCNo AS CKYC_NUMBER,CONVERT(VARCHAR(10),
SID_RESPONSEDATE, 105)  as RECEVIED_DATE , ROW_NUMBER() over (partition by CUSTOMER_REFERENCE_NUMBER order by CUSTOMER_REFERENCE_NUMBER desc) as row_num
FROM CKYC_FIN_DATA H join IF_CustomerMaster on CUSTOMER_REFERENCE_NUMBER=CM_CUST_ID
join IF_CUSTOMER_STATUS on CM_ID=ICS_CM_ID  left JOIN IF_IMMEDIATE_RESPONSE ON CM_ID=IMR_CM_ID join IF_SEARCH_IDENTITYDETAILS  on cm_id=SID_CM_ID
left JOIN IF_MASTER_RESPONSESTATUS i ON ics_Responsestatus = IRM_RESPONSECODE
where CM_CKYCNo is not null
) t where row_num = 1 order by CUSTOMER_ID,row_num desc
